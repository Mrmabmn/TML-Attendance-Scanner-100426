} } /** * Write endpoint used by the GitHub Pages scanner. * * POST parameters: * alpha_id = existing QR payload * token = random per-scan token */ function doPost(e) { const p = (e && e.parameter) || {}; } return HtmlService.createHtmlOutput( 'OK' ); } /* ========================================================================== */ /* CHECK-IN ENGINE */ /* ========================================================================== */ function checkIn_(alphaId) { const lock = LockService.getScriptLock(); } /* ========================================================================== */ /* RESULT / FORMATTING HELPERS */ /* ========================================================================== */ function result_( state, title, message, memberName, alphaId, eventName, eventId, eventDate, count, time ) { return { state: state, title: title, message: message, memberName: memberName || '', alphaId: alphaId || '', eventName: eventName || '', eventId: eventId || '', eventDate: eventDate || '', count: Number(count || 0), time: time || '' }; } function normalizeAlphaId_(value) { if ( value === null || value === undefined ) { return ''; } return String(value) .trim() .replace(/\.0$/, ''); } /** * Event dates are calendar dates, not timestamps. * * This function deliberately avoids new Date("10/10/2026"), * which can introduce UTC/local-time shifts. */ function formatEventDate_(value) { if (!value) { return '';
